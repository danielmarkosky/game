# How the player moves. Change a number, press F5 in the game, and try it: no compiling.
#
# Distances are in pixels (8 to a tile, the screen is 320 x 180), speeds in pixels a second,
# accelerations in pixels a second each second, times in seconds. Anything left out, or broken,
# is the default from src/Physics.h, and the game says what was wrong (in the corner, and in the
# terminal). levelbot reads this file too, so the room checks move the same way.

# --- Running ---
run_speed 70                # top speed on the ground
run_acceleration 300        # 70 / 300: a quarter of a second to top speed
run_deceleration 450        # letting go: a short slide to a stop (turning round brakes too)
air_acceleration 100        # a little steering: in the air you mostly go where the jump sent you
air_deceleration 0          # nothing held doesn't slow you down
fling_deceleration 140      # in the air faster than running, after letting go of a swing

# --- Jumping and falling ---
gravity 800
jump_speed 215              # jump height is jump_speed^2 / (2 * gravity): about 29 px, 3.6 tiles
jump_release_gravity 3      # letting go of jump early falls this many times faster: short hops
max_fall_speed 300
ice_acceleration 100        # on ice: slow to get going...
ice_deceleration 40         # ...and slower to stop
updraft_lift 1400           # an updraft (heat rising) pushing you up, against gravity...
updraft_speed 180           # ...up to this fast
coyote_time 0.1             # a jump still works this long after running off a ledge
jump_buffer_time 0.1        # and when pressed this long before landing
air_jump_speed 195          # the updraft feather's jump

# --- The rope ---
# It's thrown, flies out and catches or comes back; let go of it and it comes back too. It can only
# be thrown again once it's back in your hand, so the wait is however far it has to come.
rope_range 64               # how far a throw from the ground reaches, from your middle
long_rope_range 104         # with the long rope
air_throw_range 0.85        # a throw in the air reaches this much as far: nothing to push off
throw_windup 0.3            # swinging it round before throwing, until you can throw in the air
throw_speed 320             # flying out: 64 px takes 0.2 s
rope_return_speed 120       # coming back, slower: 64 px of rope takes 0.53 s, 104 px 0.87 s
swing_push 140              # holding left / right on the rope: only below where it hangs from
max_swing_speed 210
reel_speed 60               # up / down on the rope
rope_jump_speed 0           # jumping on the rope lets go of it with this much upward speed. 0: just lets go

# --- Climbing walls with your hands (the chalk bag) ---
# Push into a wall in the air to grab it; up / down climb, jump leaps off. Not slick stone.
climb_speed 40              # up
climb_down_speed 70
climb_stamina 4             # seconds of holding on, climbing up uses it twice as fast. Back on the ground
wall_slide_speed 40         # sliding down when it's gone
wall_jump_push 110          # leaping off: away from the wall this fast, and up like a jump
climb_over_speed 150        # climbing over the top: a hop up onto it

# --- Swimming ---
# In water you float with your head and shoulders out. Left / right swim, down dives, jump at the
# surface leaps out. Under the surface you hold your breath: the bar over your head.
swim_speed 45               # across, and down holding down
swim_acceleration 220
buoyancy 70                 # the lift back up to floating, for each pixel too deep: a spring
water_drag 5                # how much of your speed up and down the water takes away each second
swim_jump_speed 200         # leaping out at the surface: 25 px, up onto a ledge at the water's edge
air_time 8                  # seconds of breath under water: when it runs out you drown
air_refill_time 1           # seconds with your head out to get it all back
diving_air 4                # the diving suit (from Mags's stall): breath lasts this many times as long

# --- Bounce mushrooms ---
bounce_speed 330            # landing on one throws you up this fast: 68 px, 8.5 tiles. Hold down to stand on it

# --- The rope tied off (the knot) ---
# Hooked, and standing (or holding a wall), E ties it off where you are: it stays strung, a line to
# hang from. Grab it in the air; E again unties it, and it comes back.
line_hand_speed 45          # along it hand over hand, where it isn't steep
line_steep 0.45             # steeper than this (0 flat, 1 straight up), you slide down it...
line_zip_speed 200          # ...up to this fast
line_climb_speed 30         # or climb up it (holding up it) this fast
line_jump_speed 230         # jumping off it: enough to get onto a ledge from a line hooked at its lip

# --- Stone blocks ---
stone_drag_speed 40         # hooked, and reeled in standing (or holding a wall), the rope drags one this fast

# --- Mine carts ---
cart_speed 110              # riding one, it rolls along the rails this fast. Jump jumps the cart

# --- Floating, where there's no gravity (the Still) ---
# Out in the open nothing you do moves you: you drift on the way you were going. Touching rock you
# hold on to it, and can clamber along it, but never push off it. Only the rope gets you anywhere:
# reel in and it pulls you along it, and held, it swings you round where it's hooked, to let go of
# when you're heading the right way.
float_drag 0.1              # how much of your speed you lose a second, drifting: hardly any
float_crawl 30              # holding on to rock, the arrows move you along it, hand over hand, this fast
throw_recoil 16             # throwing the rope pushes you back the other way, this fast
space_reel_pull 320         # reeling in pulls you along the rope, this hard...
space_reel_speed 130        # ...up to this fast
